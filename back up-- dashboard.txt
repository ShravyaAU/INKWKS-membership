        function enforceView(targetRole) {
            const user = JSON.parse(localStorage.getItem('inkwks_user'));
            
            // Allow admins to go anywhere. Allow students to view workshops. Otherwise, restrict.
            if (user.role !== 'admin' && targetRole !== user.role && !(user.role === 'student' && targetRole === 'workshop')) {
                alert("Access Denied."); return;
            }

            // Fix: Force the nav bar to show on the workshop view regardless of underlying DB role
            if (targetRole === 'student' || targetRole === 'workshop') {
                document.getElementById('student-subnav').style.display = 'flex';
            } else {
                document.getElementById('student-subnav').style.display = 'none';
            }

            document.querySelectorAll('.view-section').forEach(el => el.classList.remove('active-view'));
            const badge = document.getElementById('current-view-badge');

            if (targetRole === 'student') {
                document.getElementById('student-view').classList.add('active-view');
                badge.innerText = 'STUDENT PASS';
            } else if (targetRole === 'faculty') {
                document.getElementById('faculty-view').classList.add('active-view');
                badge.innerText = 'FACULTY / STAFF';
            } else if (targetRole === 'club') {
                document.getElementById('club-view').classList.add('active-view');
                badge.innerText = 'STUDENT CLUB';
            } else if (targetRole === 'workshop') {
                document.getElementById('workshop-view').classList.add('active-view');
                badge.innerText = 'WORKSHOP PARTICIPANT';
            } else if (targetRole === 'admin') {
                document.getElementById('admin-view').classList.add('active-view');
                badge.innerText = 'ADMIN PORTAL';
            }
            closeRoleModal();
        }

        async function fetchPassStatus(userId) {
            try {
                const response = await fetch(`http://127.0.0.1:8000/api/memberships/${userId}`);
                const data = await response.json();
                const badgeElement = document.getElementById('pass-status-badge');
                if (data.has_pass) {
                    badgeElement.innerText = `ACTIVE (${data.sessions_remaining} LEFT)`;
                    badgeElement.classList.remove('status-none');
                    badgeElement.style.backgroundColor = 'rgba(255,198,39,0.1)';
                    badgeElement.style.color = 'var(--asu-gold)';
                    badgeElement.style.borderColor = 'var(--asu-gold)';
                }
                const aptResponse = await fetch(`http://127.0.0.1:8000/api/appointments/${userId}`);
                if (aptResponse.ok) {
                    const appointments = await aptResponse.json();
                    if (appointments.length > 0) {
                        document.getElementById('appointment-list').innerHTML = appointments.map(apt =>

            const title = document.getElementById('screen-title').value;
            const mesh = document.getElementById('screen-mesh').value;
            const user = JSON.parse(localStorage.getItem('inkwks_user'));

        async function manageSession(action) {
            const email = document.getElementById('session-target-email').value;
            if (!email) {
                alert("Please enter the student's email.");
                return;
            }
            
            const user = JSON.parse(localStorage.getItem('inkwks_user'));

            try {
                const response = await fetch('http://127.0.0.1:8000/api/admin/sessions', {
                    method: 'PUT',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        admin_id: user.id,
                        target_email: email,
                        action: action // "deduct" or "restore"
                    })
                });

                const data = await response.json();

                if (response.ok) {
                    alert(`Success: ${data.message}\nSessions Remaining: ${data.sessions_remaining}`);
                    document.getElementById('session-target-email').value = ''; // clear the input
                } else {
                    alert(`Error: ${data.detail}`);
                }
            } catch (error) {
                console.error("Session management error:", error);
                alert("Connection error to server. Is your FastAPI backend running?");
            }
        }

        async function confirmBooking() {
            const aptTime = document.getElementById('appointment-time').value;
            if (!aptTime) {
                document.getElementById('booking-error').innerText = "Please select a date and time.";
                return;
            }
            
            const user = JSON.parse(localStorage.getItem('inkwks_user'));
            
            try {
                const response = await fetch('http://127.0.0.1:8000/api/appointments', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        user_id: user.id,
                        appointment_time: aptTime
                    })
                });
                
                const data = await response.json();
                
                if (response.ok) {
                    alert("Session booked successfully!");
                    closeBookingModal();
                    fetchPassStatus(user.id); // Instantly refresh the UI list
                } else {
                    document.getElementById('booking-error').innerText = data.detail || "Error booking session. Do you have an active pass?";

                console.error("Booking error:", error);
                document.getElementById('booking-error').innerText = "Connection error to server.";
            }
        }
        
        async function createWorkshop() {
            const title = document.getElementById('new-ws-title').value;
            const instructor = document.getElementById('new-ws-instructor').value;
            const price = document.getElementById('new-ws-price').value;
            const capacity = document.getElementById('new-ws-capacity').value;
            const date = document.getElementById('new-ws-date').value;

            if (!title || !price || !capacity || !date) {
                alert("Please fill in the title, price, capacity, and date."); return;
            }

            const user = JSON.parse(localStorage.getItem('inkwks_user'));

            try {
                const response = await fetch('http://127.0.0.1:8000/api/admin/workshops', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        admin_id: user.id,
                        title: title,
                        instructor: instructor || "TBA",
                        price: parseFloat(price),
                        available_seats: parseInt(capacity),
                        workshop_date: date
                    })
                });

                if (response.ok) {
                    alert("Workshop published successfully!");
                    // Clear the form fields
                    document.getElementById('new-ws-title').value = '';
                    document.getElementById('new-ws-instructor').value = '';
                    document.getElementById('new-ws-price').value = '';
                    document.getElementById('new-ws-capacity').value = '';
                    document.getElementById('new-ws-date').value = '';
                    
                    // Refresh the live admin list
                    loadAdminData(user.id); 
                } else {
                    const error = await response.json();
                    alert(`Error: ${error.detail}`);
                }
            } catch (error) {
                console.error("Error creating workshop:", error);
                alert("Connection error to server. Is FastAPI running?");
            }
        }

        async function deleteWorkshop(workshopId) {
            if(!confirm("Are you sure you want to permanently delete this workshop? This will also remove any student registrations for it.")) return;
            
            const user = JSON.parse(localStorage.getItem('inkwks_user'));
            
            try {
                // Pass the admin ID in the URL for security verification
                const response = await fetch(`http://127.0.0.1:8000/api/workshops/${workshopId}?admin_id=${user.id}`, {
                    method: 'DELETE'
                });
                
                if (response.ok) {
                    alert("Workshop deleted successfully.");
                    loadAdminData(user.id); // Refresh admin list
                    fetchWorkshopData(user.id); // Refresh public student list
                } else {
                    const data = await response.json();
                    alert(`Error: ${data.detail}`);
                }
            } catch (e) {
                alert("Connection error to server.");
            }
        }

        async function loadAdminData(adminId) {
            // 1. Fetch Screens
            try {
                const res = await fetch(`http://127.0.0.1:8000/api/admin/screens?admin_id=${adminId}`);
                if (res.ok) {
                    const screens = await res.json();
                    document.getElementById('admin-screen-list').innerHTML = screens.length > 0 ? 
                        screens.map(s => 
                            `<div style="padding: 15px; background-color: #222; border-radius: 8px; margin-bottom: 10px; display: flex; justify-content: space-between; align-items: center; border-left: 4px solid #EF4444;">
                                <div><strong style="color: white;">${s.title}</strong><div style="color: var(--asu-gold); font-size: 12px; margin-top: 4px;">User: ${s.email}</div></div>
                                <div style="display: flex; gap: 10px; align-items: center;">
                                    <span style="color: #10B981; font-size: 12px; font-weight: bold; text-transform: uppercase;">${s.status}</span>
                                    <select onchange="updateScreenStatus('${s.id}', this.value)" style="padding: 6px; background: #333; border: 1px solid #555; color: white; width: auto; margin: 0;"><option value="">Change...</option><option value="ready">Ready</option><option value="collected">Collected</option></select>
                                </div>
                            </div>`
                        ).join('') : "No screens currently in the pipeline.";
                }
            } catch (error) { console.error(error); }

            // 2. Fetch Workshops
            try {
                const wsRes = await fetch('http://127.0.0.1:8000/api/workshops/available');
                if (wsRes.ok) {
                    const workshops = await wsRes.json();
                    document.getElementById('admin-active-workshops').innerHTML = workshops.length > 0 ? 
                        workshops.map(w => 
                            `<div style="padding: 10px; background-color: #222; border-radius: 6px; border-left: 4px solid #8B5CF6; color: white; margin-bottom: 10px; display: flex; justify-content: space-between; align-items: center;">
                                <div><strong>${w.title}</strong> <br><span style="font-size: 12px; color: var(--text-muted);">${w.available_seats} seats remaining</span></div>
                                <div style="display: flex; gap: 5px;">
                                    <button onclick="viewRoster('workshop', '${w.id}')" style="background-color: #333; color: white; border: none; padding: 5px 10px; border-radius: 4px; cursor: pointer; font-size: 12px;">Roster</button>
                                    <button onclick="deleteWorkshop('${w.id}')" style="background-color: rgba(239,68,68,0.2); color: #EF4444; border: 1px solid #EF4444; padding: 5px 10px; border-radius: 4px; cursor: pointer; font-size: 12px;">Delete</button>
                                </div>
                            </div>`
                        ).join('') : "No active workshops.";
                }
            } catch (e) { console.error(e); }

            // 3. Fetch LIVE Schedule
            try {
                const schedRes = await fetch(`http://127.0.0.1:8000/api/admin/schedule?admin_id=${adminId}`);
                if (schedRes.ok) {
                    const schedule = await schedRes.json();
                    document.getElementById('admin-schedule-list').innerHTML = schedule.length > 0 ? 
                        schedule.map(s => 
                            `<div style="padding: 10px; background-color: #222; border-radius: 6px; border-left: 4px solid var(--asu-gold); color: white; margin-bottom: 10px; display: flex; justify-content: space-between; align-items: center;">
                                <div><strong>Student:</strong> ${s.email} <br><span style="font-size: 12px; color: var(--text-muted);">${new Date(s.appointment_time).toLocaleString()}</span></div>
                            </div>`
                        ).join('') : "No upcoming sessions booked.";
                }
            } catch (e) { console.error(e); }

            // 4. Fetch LIVE Inbox
            try {
                const inboxRes = await fetch(`http://127.0.0.1:8000/api/admin/inbox?admin_id=${adminId}`);
                if (inboxRes.ok) {
                    const inbox = await inboxRes.json();
                    const allRequests = [...inbox.faculty, ...inbox.clubs];
                    
                    document.getElementById('admin-inbox-list').innerHTML = allRequests.length > 0 ? 
                        allRequests.map(r => 
                            `<div style="padding: 10px; background-color: #222; border-radius: 6px; border-left: 4px solid ${r.type === 'faculty' ? '#8B5CF6' : '#10B981'}; color: white; margin-bottom: 10px; display: flex; justify-content: space-between; align-items: center;">
                                <div><strong style="text-transform: capitalize;">${r.type}:</strong> ${r.title} <br><span style="font-size: 12px; color: var(--text-muted);">Requested: ${r.participants} participants</span></div>
                                <button onclick="reviewRequest('${r.type}', '${r.id}')" style="background-color: #10B981; color: white; border: none; padding: 5px 10px; border-radius: 4px; cursor: pointer;">Approve</button>
                            </div>`
                        ).join('') : "No pending requests.";
                }
            } catch (e) { console.error(e); }
        }

        async function reviewRequest(type, id) {
            if(!confirm("Mark this request as Approved?")) return;
            const user = JSON.parse(localStorage.getItem('inkwks_user'));
            
            try {
                const response = await fetch('http://127.0.0.1:8000/api/admin/inbox/review', {
                    method: 'PUT',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ admin_id: user.id, request_type: type, request_id: id, new_status: 'approved' })
                });
                
                if (response.ok) {
                    alert("Request approved successfully.");
                    loadAdminData(user.id); // Instantly removes it from the inbox
                } else {
                    alert("Failed to update request.");
                }
            } catch (e) { console.error(e); }
        }

        async function viewRoster(type, id) {
            const user = JSON.parse(localStorage.getItem('inkwks_user'));
            try {
                const response = await fetch(`http://127.0.0.1:8000/api/admin/roster/${type}/${id}?admin_id=${user.id}`);
                if (response.ok) {
                    const roster = await response.json();
                    if (roster.length > 0) {
                        alert(`REGISTERED STUDENTS:\n\n${roster.join('\n')}`);
                    } else {
                        alert("No students are currently registered for this.");
                    }
                }
            } catch (e) { console.error(e); }
        }
