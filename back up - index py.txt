# Database connection generator using your Neon URL
def get_db():
    conn = psycopg2.connect(
        os.getenv("NEON_DATABASE_URL"),
        cursor_factory=RealDictCursor
    )
    try:
        yield conn
    finally:
        conn.close()

@app.post("/api/auth/google")
async def verify_google_sso(token_data: GoogleToken, db=Depends(get_db)):
    # 1. Verify the token with Google's public endpoint
    async with httpx.AsyncClient() as client:
        response = await client.get(
            f"https://oauth2.googleapis.com/tokeninfo?id_token={token_data.token}"
        )
    
    if response.status_code != 200:
        raise HTTPException(status_code=401, detail="Invalid Google token")
        
    user_info = response.json()
    email = user_info.get("email", "")
    
    # 2. Enforce strict university access
    if not email.endswith("@asu.edu"):
        raise HTTPException(status_code=403, detail="Only @asu.edu emails are permitted.")
        
    cursor = db.cursor()
    
    # 3. Check if user exists, or create a new profile automatically
    cursor.execute("SELECT id, role, orientation_completed FROM users WHERE email = %s", (email,))
    user = cursor.fetchone()
    
    if not user:
        cursor.execute(
            "INSERT INTO users (email, role) VALUES (%s, 'student') RETURNING id, role, orientation_completed", 
            (email,)
        )
        user = cursor.fetchone()
        db.commit()
     
    # the frontend receives the email   
    user["email"] = email
        
    return {
        "message": "Authentication successful",
        "user": user
    }

@app.get("/api/memberships/{user_id}")
def get_membership_status(user_id: str, db=Depends(get_db)):
    cursor = db.cursor()
    # Find the most recent active pass that hasn't expired
    cursor.execute("""
        SELECT sessions_remaining, expiration_date 
        FROM memberships 
        WHERE user_id = %s AND status = 'active' AND expiration_date > CURRENT_TIMESTAMP
        ORDER BY expiration_date DESC LIMIT 1
    """, (user_id,))
    
    membership = cursor.fetchone()
    
    if membership:
        return {"has_pass": True, "sessions_remaining": membership["sessions_remaining"]}
    
    return {"has_pass": False, "sessions_remaining": 0}

class AppointmentRequest(BaseModel):
    user_id: str
    appointment_time: str

@app.post("/api/appointments")
def book_appointment(req: AppointmentRequest, db=Depends(get_db)):
    cursor = db.cursor()
    
    # 1. Verify the user has a valid pass with remaining sessions
    cursor.execute("""
        SELECT id, sessions_remaining 
        FROM memberships 
        WHERE user_id = %s AND status = 'active' AND expiration_date > CURRENT_TIMESTAMP
        ORDER BY expiration_date DESC LIMIT 1
    """, (req.user_id,))
    membership = cursor.fetchone()
    
    if not membership or membership['sessions_remaining'] <= 0:
        raise HTTPException(status_code=400, detail="No active sessions remaining on your pass.")
        
    # 2. Book the appointment
    cursor.execute("""
        INSERT INTO appointments (user_id, appointment_time, status)
        VALUES (%s, %s, 'booked')
    """, (req.user_id, req.appointment_time))
    
    # 3. Deduct one session from the active membership
    cursor.execute("""
        UPDATE memberships 
        SET sessions_remaining = sessions_remaining - 1 
        WHERE id = %s
        RETURNING sessions_remaining
    """, (membership['id'],))
    
    updated_sessions = cursor.fetchone()['sessions_remaining']
    db.commit()
    
    return {
        "message": "Appointment booked successfully!", 
        "sessions_remaining": updated_sessions
    }

@app.put("/api/admin/roles")
def update_user_role(req: RoleUpdateRequest, db=Depends(get_db)):
    cursor = db.cursor()
    
    # 1. Verify the requester is actually an Admin
    cursor.execute("SELECT role FROM users WHERE id = %s", (req.admin_id,))
    requester = cursor.fetchone()
    if not requester or requester['role'] != 'admin':
        raise HTTPException(status_code=403, detail="Unauthorized: Admin access required.")
        
    # 2. Verify the target role is valid based on your Enums
    valid_roles = ['student', 'faculty', 'club', 'workshop', 'admin']
    if req.new_role not in valid_roles:
        raise HTTPException(status_code=400, detail="Invalid role specified.")
        
    # 3. Update the target user
    cursor.execute("UPDATE users SET role = %s WHERE email = %s RETURNING id", (req.new_role, req.target_email))
    updated_user = cursor.fetchone()
    
    if not updated_user:
        raise HTTPException(status_code=404, detail="User not found.")
        
    db.commit()
    return {"message": f"Successfully updated {req.target_email} to {req.new_role}."}

class WorkshopRegistration(BaseModel):
    user_id: str
    workshop_id: str

@app.post("/api/workshops/register")
def register_workshop(req: WorkshopRegistration, db=Depends(get_db)):
    cursor = db.cursor()
    
    # 1. Check if seats are still available
    cursor.execute("SELECT available_seats FROM workshops WHERE id = %s", (req.workshop_id,))
    workshop = cursor.fetchone()
    
    if not workshop or workshop['available_seats'] <= 0:
        raise HTTPException(status_code=400, detail="Sorry, this workshop is full.")
        
    # 2. Register the user
    cursor.execute("INSERT INTO workshop_registrations (workshop_id, user_id) VALUES (%s, %s)", (req.workshop_id, req.user_id))
    
    # 3. Deduct one available seat
    cursor.execute("UPDATE workshops SET available_seats = available_seats - 1 WHERE id = %s", (req.workshop_id,))
    
    db.commit()
    return {"message": "Successfully registered for the workshop!"}

# --- NEW: Admin Session Manager Endpoint ---
@app.put("/api/admin/sessions")
def update_session_count(req: SessionUpdate, db=Depends(get_db)):
    cursor = db.cursor()
    
    try:
        # 1. Verify the requester is an Admin
        cursor.execute("SELECT role FROM users WHERE id = %s", (req.admin_id,))
        requester = cursor.fetchone()
        if not requester or requester['role'] != 'admin':
            raise HTTPException(status_code=403, detail="Unauthorized: Admin access required.")

        # 2. Find the target user by email
        cursor.execute("SELECT id FROM users WHERE email = %s", (req.target_email,))
        target_user = cursor.fetchone()
        
        if not target_user:
            raise HTTPException(status_code=404, detail="Student email not found.")
            
        # 3. Find their active membership
        cursor.execute("""
            SELECT id, sessions_remaining 
            FROM memberships 
            WHERE user_id = %s AND status = 'active'
        """, (target_user['id'],))
        membership = cursor.fetchone()
        
        if not membership:
            raise HTTPException(status_code=400, detail="Student does not have an active print pass.")
            
        current_sessions = membership['sessions_remaining']
        
        # 4. Apply the logic (Deduct or Restore)
        if req.action == "deduct":
            if current_sessions <= 0:
                raise HTTPException(status_code=400, detail="Student has 0 sessions left.")
            new_sessions = current_sessions - 1
        elif req.action == "restore":
            new_sessions = current_sessions + 1
        else:
            raise HTTPException(status_code=400, detail="Invalid action. Use 'deduct' or 'restore'.")
            
        # 5. Update the database
        cursor.execute("""
            UPDATE memberships 
            SET sessions_remaining = %s 
            WHERE id = %s
        """, (new_sessions, membership['id']))
        
        db.commit()
        
        return {
            "message": f"Successfully {req.action}ed session for {req.target_email}.",
            "sessions_remaining": new_sessions
        }
        
    except HTTPException:
        # Re-raise standard HTTP exceptions
        raise
    except Exception as e:
        db.rollback()
        raise HTTPException(status_code=500, detail=str(e))

        raise HTTPException(status_code=403, detail="Unauthorized: Admin access required.")

    except Exception as e:
        db.rollback()
        raise HTTPException(status_code=500, detail=str(e))

class RequestReview(BaseModel):

    request_id: str
    new_status: str

@app.put("/api/admin/inbox/review")
def review_inbox_request(req: RequestReview, db=Depends(get_db)):
    cursor = db.cursor()
    cursor.execute("SELECT role FROM users WHERE id = %s", (req.admin_id,))
    if cursor.fetchone().get('role') != 'admin':
        raise HTTPException(status_code=403, detail="Unauthorized")
        
    if req.request_type == 'faculty':
        cursor.execute("UPDATE faculty_requests SET status = %s WHERE id = %s", (req.new_status, req.request_id))
    elif req.request_type == 'club':
        cursor.execute("UPDATE club_requests SET status = %s WHERE id = %s", (req.new_status, req.request_id))
        
    db.commit()
    return {"message": f"{req.request_type.capitalize()} request marked as {req.new_status}."}

@app.get("/api/admin/roster/{item_type}/{item_id}")
def get_roster(item_type: str, item_id: str, admin_id: str, db=Depends(get_db)):
    cursor = db.cursor()
    cursor.execute("SELECT role FROM users WHERE id = %s", (admin_id,))
    if cursor.fetchone().get('role') != 'admin':
        raise HTTPException(status_code=403, detail="Unauthorized")
        
    if item_type == "workshop":
        cursor.execute("""
            SELECT u.email FROM workshop_registrations r
            JOIN users u ON r.user_id = u.id
            WHERE r.workshop_id = %s
        """, (item_id,))
    else:
        # For sessions, just return the specific user who booked it
        cursor.execute("""
            SELECT u.email FROM appointments a
            JOIN users u ON a.user_id = u.id
            WHERE a.id = %s
        """, (item_id,))
        
    roster = cursor.fetchall()
    return [r['email'] for r in roster]
